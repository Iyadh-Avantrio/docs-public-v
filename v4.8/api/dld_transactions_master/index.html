    Transactions Master

    This module handles loading, processing, merging, and saving of real estate transactions, mortgages, properties, and service charge data.

    Key Features¶

    • Loads environment variables and connects to development and production MongoDB instances.
    • Loads initial datasets from MongoDB collections into pandas DataFrames.
    • Calculates resale metrics such as property resale count and price changes.
    • Merges transactions with property and service charge data using multiple fallback strategies.
    • Creates detailed dictionaries of units, villas, and lands per transaction.
    • Matches mortgage data to transactions with date tolerance and fallback keys.
    • Generates building presence maps for property type assignment.
    • Merges DIFC transaction data with primary datasets.
    • Assigns unified property types for transactions and rent contracts.
    • Performs final data cleaning and preparation for database upload.
    • Saves processed data atomically to MongoDB collections with indexing.
    • Provides utility functions for data adjustments and dictionary cleaning.

    Usage¶

    Designed to produce enriched and integrated datasets for real estate analytics by combining transactional, mortgage, property, and service charge data consistently and efficiently.

    def assign_property_type_prop_ai_for_rent(df):
    """
    Assign property type 'property_type_prop-ai' for rent contracts,
    based on ejari property and usage types.

    Args:
        df (pd.DataFrame): Rent contracts dataframe.

    Returns:
        pd.DataFrame: Dataframe with new 'property_type_prop-ai' column.
    """
    def classify(row):
        sub_type = row.get("ejari_property_type_en") 
        type_en = row.get("ejari_bus_property_type_en") 
        property_usage_en = row.get("property_usage_en") 

        if property_usage_en == "Residential" and type_en == "Unit":
            return "Apartment"

        if property_usage_en == "Residential" and type_en == "Villa":
            return "Villa"

        if sub_type == "Office":
            return "Office"

        if type_en != "Residential" and sub_type == "Shop":
            return "Shop"

        if type_en != "Residential" and sub_type == "Office":
            return "Office"

        if sub_type == "Hotel apartments":
            return "Hotel Apartment"

        if property_usage_en == "Residential" and type_en == "Unit" and sub_type == "Hotel":
            return "Hotel Apartment"

        return type_en


    df["property_type_prop-ai"] = df.apply(classify, axis=1)
    return df

    def assign_property_type_prop_ai_for_transactions(df, building_check_set):
    """
    Assign a unified property type 'property_type_prop-ai' for transactions
    based on subtypes, main types, and building presence.

    Args:
        df (pd.DataFrame): Transactions dataframe.
        building_check_set (set): Set of project_numbers identified as buildings.

    Returns:
        pd.DataFrame: Dataframe with new 'property_type_prop-ai' column.
    """
    def classify(row):
        # Use ejari values if available, otherwise fallback to main ones
        sub_type = row.get("property_sub_type_en")
        type_en = row.get("property_type_en")
        proj_num = row.get("project_number")

        if sub_type == "Flat":
            return "Apartment"
        if sub_type == "Office":
            return "Office"
        if sub_type == "Shop":
            return "Shop"
        if sub_type in {"Hotel Apartment", "Hotel Room", "Hotel apartments"}:
            return "Hotel Apartment"

        if type_en == "Villa":
            return "Villa"

        if type_en == "Land" and pd.notna(proj_num):
            if proj_num not in building_check_set:
                return "Villa (Land)"
            else:
                return "Land"

        return type_en

    df["property_type_prop-ai"] = df.apply(classify, axis=1)
    return df

    def calculate_resale_metrics(transactions_df):
    """
    Calculate resale metrics for property transactions.

    This includes calculating:
    - Number of times a property was sold.
    - Price change relative to the last sale.
    - Price change relative to the first sale of the property.

    Args:
        transactions_df (pandas.DataFrame): DataFrame containing property transaction records.

    Returns:
        pandas.DataFrame: Transactions DataFrame enriched with resale metrics.

    Logs:
        Number of CPU cores used for parallel processing.
        Progress of resale metrics calculation.
        Completion confirmation.
    """
    logger.info("Calculating resale metrics")

    # Preprocess
    transactions_df["instance_date"] = pd.to_datetime(transactions_df["instance_date"])
    transactions_df = transactions_df.sort_values(['property_id', 'instance_date']).reset_index(drop=True)

    def calculate_metrics_single_group(group):
        group = group.sort_values('instance_date').copy()
        group['number_times_sold'] = range(1, len(group) + 1)
        group['last_worth'] = group['actual_worth'].shift(1)
        group['price_change_vs_last_sale'] = (
            (group['actual_worth'] / group['last_worth']) - 1
        ).where(group['last_worth'].notna() & (group['last_worth'] != 0))

        try:
            first_worth = group.loc[group['sale_type_id'] == 1, 'actual_worth'].iloc[0]
            if first_worth != 0:
                group['price_change_vs_first_sale'] = (group['actual_worth'] / first_worth) - 1
            else:
                group['price_change_vs_first_sale'] = pd.NA
        except IndexError:
            group['price_change_vs_first_sale'] = pd.NA

        return group.drop(columns='last_worth')

    # Parallel processing
    grouped = list(transactions_df.groupby('property_id'))
    num_cores = multiprocessing.cpu_count()
    logger.info(f"Processing resale metrics with {num_cores} cores")

    results = Parallel(n_jobs=num_cores)(
        delayed(calculate_metrics_single_group)(group) 
        for _, group in tqdm(grouped, desc="Processing resale metrics")
    )

    # Batch concatenation
    batch_size = 2000
    batches = []
    for i in tqdm(range(0, len(results), batch_size), desc="Concatenating batches"):
        batch = pd.concat(results[i:i + batch_size], ignore_index=True)
        batches.append(batch)

    final_df = pd.concat(batches, ignore_index=True)
    logger.info("Resale metrics calculation completed")
    return final_df

    def clean_data(df):
    """
    Final cleaning of the dataframe:
    - Replace NaN values with empty strings in dicts and fields.
    - Reset unit, villa, land numbers to empty dict if project_number is missing.
    - Convert 'sale_type_id' to numeric if present.

    Args:
        df (pd.DataFrame): Input dataframe to clean.

    Returns:
        pd.DataFrame: Cleaned dataframe.
    """
    logger.info("Performing final data cleaning")

    def replace_nan_with_empty_string(obj):
        if pd.isna(obj):
            return ""
        if isinstance(obj, dict):
            new_dict = {}
            for k, v in obj.items():
                new_k = "" if pd.isna(k) else k
                if isinstance(v, dict):
                    new_v = replace_nan_with_empty_string(v)
                elif pd.isna(v):
                    new_v = ""
                else:
                    new_v = v
                new_dict[new_k] = new_v
            return new_dict
        return obj


    # If project_number is NA, set unit/villa/land numbers to empty dict
    df.loc[df['project_number'].isna(), ['unit_numbers_all', 'villa_numbers_all', 'land_numbers_all', 'unit_numbers', 'villa_numbers', 'land_numbers']] = [{}] * 6

    # Apply cleaning to individual fields
    for col in ['unit_numbers_all', 'villa_numbers_all', 'land_numbers_all', 'unit_numbers', 'villa_numbers', 'land_numbers']:
        if col in df.columns:
            df[col] = df[col].apply(replace_nan_with_empty_string)

    logger.info("Data cleaning completed")


    if 'sale_type_id' in df.columns:
        df["sale_type_id"] = pd.to_numeric(df["sale_type_id"], errors='coerce')

    return df

    def create_units_and_villas(transactions_df, property_details_df):
    """
    Create dictionaries of units, villas, and lands for each transaction in the dataset.

    This function matches transactions with property details based on composite keys formed
    from project number, area, rooms, and property subtype. It separates properties into units,
    villas, and lands, and aggregates their relevant details into dictionaries attached to each transaction.

    Args:
        transactions_df (pd.DataFrame): DataFrame of property transactions.
        property_details_df (pd.DataFrame): DataFrame with detailed property attributes.

    Returns:
        pd.DataFrame: transactions_df enriched with columns:
            - 'unit_numbers'
            - 'villa_numbers'
            - 'land_numbers'
        Each containing a dictionary of relevant property details keyed by unit/building number.

    Logs:
        Info on the creation process start and successful completion.
    """
    logger.info("Creating project dictionaries")

    # Ensure unique indices to avoid reindexing errors
    transactions_df = transactions_df.reset_index(drop=True)

    property_details_df = property_details_df[~property_details_df["project_number"].isna()]

    # Convert to numeric once
    transactions_df['project_number_numeric'] = pd.to_numeric(transactions_df['project_number'], errors='coerce')
    property_details_df['project_number_numeric'] = pd.to_numeric(property_details_df['project_number'], errors='coerce')

    # Create merge keys for efficient joining
    transactions_df['merge_key'] = (
        transactions_df['project_number_numeric'].astype(str) + '_' +
        transactions_df['procedure_area'].astype(str) + '_' +
        transactions_df['rooms_en'].astype(str) + '_' +
        transactions_df['property_sub_type_en'].astype(str)
    )

    property_details_df['merge_key'] = (
        property_details_df['project_number_numeric'].astype(str) + '_' +
        property_details_df['actual_area'].astype(str) + '_' +
        property_details_df['rooms_en'].astype(str) + '_' +
        property_details_df['property_sub_type_en'].astype(str)
    )

    # Initialize result columns
    transactions_df['unit_numbers_all'] = [{}] * len(transactions_df)
    transactions_df['villa_numbers_all'] = [{}] * len(transactions_df)
    transactions_df['land_numbers_all'] = [{}] * len(transactions_df)

    # Separate property details by source type for efficiency
    unit_details = property_details_df[property_details_df['Source_Data'] == 'dld_units'].copy()
    land_details = property_details_df[property_details_df['Source_Data'] == 'dld_lands'].copy()

    # Process Units (Apartments & Hotel Apartments)
    apartment_mask = transactions_df['property_type_prop-ai'].isin(['Apartment', 'Hotel Apartment'])

    if apartment_mask.any() and not unit_details.empty:
        # Group unit details by merge_key
        unit_groups = unit_details.groupby('merge_key').apply(
            lambda group: {
                row['unit_number']: {
                    'floor': row['floor'],
                    'unit_balcony_area': row['unit_balcony_area'],
                    'unit_parking_number': row['unit_parking_number'], 
                    'rooms': row['rooms'],
                    'rooms_en': row['rooms_en'], 
                    'building_number': row['building_number'],
                    'property_type_en': row['property_type_en'], 
                    'property_sub_type_en': row['property_sub_type_en'],
                    'land_type_en': row['land_type_en'], 
                    'parcel_id': row['parcel_id'], 
                    'property_id': row['property_id']
                }
                for _, row in group.iterrows()
            }
        ).to_dict()

        # Map unit dictionaries to transactions using boolean indexing
        transactions_df.loc[apartment_mask, 'unit_numbers_all'] = (
            transactions_df.loc[apartment_mask, 'merge_key']
            .map(lambda key: unit_groups.get(key, {}))
        )

    # Process Villas
    villa_mask = transactions_df['property_type_prop-ai'] == 'Villa'

    if villa_mask.any() and not unit_details.empty:
        # Group villa details by merge_key
        villa_groups = unit_details.groupby('merge_key').apply(
            lambda group: {
                row['building_number']: {
                    'built_up_area': row['built_up_area'], 
                    'car_parks': row['car_parks'],
                    'rooms': row['rooms'], 
                    'rooms_en': row['rooms_en'], 
                    'floors': row['floors'],
                    'property_type_en': row['property_type_en'], 
                    'property_sub_type_en': row['property_sub_type_en'],
                    'land_type_en': row['land_type_en'], 
                    'parcel_id': row['parcel_id'], 
                    'property_id': row['property_id']
                }
                for _, row in group.iterrows()
            }
        ).to_dict()

        # Map villa dictionaries to transactions using boolean indexing
        transactions_df.loc[villa_mask, 'villa_numbers_all'] = (
            transactions_df.loc[villa_mask, 'merge_key']
            .map(lambda key: villa_groups.get(key, {}))
        )

    # Process Lands
    land_mask = transactions_df['property_type_prop-ai'].isin(['Villa (Land)', 'Land'])

    if land_mask.any() and not land_details.empty:
        # Group land details by merge_key
        land_groups = land_details.groupby('merge_key').apply(
            lambda group: {
                row['building_number']: {
                    'land_type_en': row['land_type_en'], 
                    'parcel_id': row['parcel_id'], 
                    'property_id': row['property_id'],
                    'actual_area': row['actual_area']
                }
                for _, row in group.iterrows()
            }
        ).to_dict()

        # Map land dictionaries to transactions using boolean indexing
        transactions_df.loc[land_mask, 'land_numbers_all'] = (
            transactions_df.loc[land_mask, 'merge_key']
            .map(lambda key: land_groups.get(key, {}))
        )

    def clear_if_multiple_keys(d):
        """Clear dictionary if it contains more than 1 key"""
        return {} if isinstance(d, dict) and len(d) > 1 else d

    def clean_dict(d):
        return {} if not isinstance(d, dict) else d

    transactions_df['unit_numbers_all'] = transactions_df['unit_numbers_all'].apply(clean_dict)
    transactions_df['villa_numbers_all'] = transactions_df['villa_numbers_all'].apply(clean_dict)
    transactions_df['land_numbers_all'] = transactions_df['land_numbers_all'].apply(clean_dict)

    # Apply clear_if_multiple_keys to create the final columns
    transactions_df['unit_numbers'] = transactions_df['unit_numbers_all'].apply(clear_if_multiple_keys)
    transactions_df['villa_numbers'] = transactions_df['villa_numbers_all'].apply(clear_if_multiple_keys)
    transactions_df['land_numbers'] = transactions_df['land_numbers_all'].apply(clear_if_multiple_keys)

    # Clean up temporary columns
    transactions_df.drop(['project_number_numeric', 'merge_key'], axis=1, inplace=True, errors='ignore')

    logger.info("Project dictionaries created successfully")

    return transactions_df

    def ensure_indexes(collection, fields):
    """
    Ensure single-field ascending indexes exist on specified fields.

    Args:
        collection (pymongo.collection.Collection): MongoDB collection to index.
        fields (list[str]): List of field names to index.
    """
    existing_indexes = collection.index_information()
    print(existing_indexes)
    existing_fields = {tuple(i['key'])[0][0] for i in existing_indexes.values() if 'key' in i}

    for field in fields:
        if field not in existing_fields:
            print(f"Creating index for: {field}")
            collection.remove_index
            collection.create_index([(field, 1)])

    def merge_properties_and_service_charges(transactions_df, mortgages_df, properties_df, charges_df):
    """
    Merge transaction and mortgage data with property and service charge information.

    Performs multi-step merging:
    1. Merges transactions with property details.
    2. Merges with service charges using property group ID and property usage.
    3. Falls back to merging on project number for unmatched records.
    4. Further fallback merging on prop-ai location key building for remaining unmatched.

    Args:
        transactions_df (pandas.DataFrame): Transactions data.
        mortgages_df (pandas.DataFrame): Mortgages data.
        properties_df (pandas.DataFrame): Properties details data.
        charges_df (pandas.DataFrame): Service charges data.

    Returns:
        tuple: (transactions_final, mortgages_final) DataFrames after merges.

    Logs:
        Progress of merging steps, number of duplicates removed,
        number of successful fallback merges, and final merge summary.
    """
    logger.info("Merging additional data sources")

    # Clean MongoDB _id fields
    for df in [mortgages_df, charges_df, properties_df]:
        df.drop(columns=['_id'], inplace=True, errors='ignore')

    def merge_properties_and_charges(df):
        # Properties fields
        props_fields = [
            "transaction_id", "transaction_size", "transaction_date", "transaction_amount",
            "sale_type_id", 
            # "property_id", 
            "property_group_id", "property_usage_id_dld",
            "property_usage_en_dld", "building_name_en_dld"
        ]

        # Drop overlapping fields except transaction_id
        columns_to_drop = [col for col in props_fields if col != "transaction_id" and col in df.columns]
        df_cleaned = df.drop(columns=columns_to_drop)

        # Merge properties
        props_trimmed = properties_df[props_fields].drop_duplicates(subset=['transaction_id'])
        df_merged = pd.merge(df_cleaned, props_trimmed, on="transaction_id", how="left")

        # Service charges fields
        charges_fields = [
            "master_project_id", "property_group_name_en", "service_charge_group_name_en",
            "property_group_id", "service_charges_budget_year", "service_charge_group_usage_en",
            "service_charges_general_fund", "service_charges_total"
        ]

        # Drop overlapping charges fields except property_group_id
        columns_to_drop = [col for col in charges_fields if col != "property_group_id" and col in df_merged.columns]
        df_merged_cleaned = df_merged.drop(columns=columns_to_drop)

        # Step 1: Primary merge using property_group_id
        charges_trimmed = charges_df[charges_fields].drop_duplicates(subset=['property_group_id', 'service_charge_group_usage_en'])
        df_with_charges = pd.merge(
            df_merged_cleaned,
            charges_trimmed,
            left_on=["property_group_id", "property_usage_en_dld"],
            right_on=["property_group_id", "service_charge_group_usage_en"],
            how="left"
        )

        display(df_with_charges)

        # Step 2: Fallback merge using project_number
        no_charges_mask = df_with_charges['service_charges_total'].isna()

        if no_charges_mask.sum() > 0 and "project_number" in charges_df.columns:
            logger.info(f"Step 2: Attempting project_number fallback merge for {no_charges_mask.sum()} records")

            # Prepare project_number fallback merge
            project_fallback_fields = [
                "project_number", "service_charge_group_name_en", "service_charges_budget_year", 
                "service_charge_group_usage_en", "service_charges_general_fund", "service_charges_total"
            ]
            available_project_fields = [field for field in project_fallback_fields if field in charges_df.columns]

            # Create project_number charges dataset
            charges_project = charges_df[available_project_fields].copy()

            # Remove rows where project_number is NaN
            charges_project = charges_project.dropna(subset=["project_number"])
            duplicate_mask = charges_project.duplicated(subset=["project_number"], keep=False)
            charges_project_unique = charges_project[~duplicate_mask]

            # Remove duplicates - if multiple matches exist, ignore them
            duplicate_mask = charges_project.duplicated(subset=["project_number"], keep=False)
            charges_project_unique = charges_project[~duplicate_mask]

            # Get records that need fallback merge
            df_no_charges = df_with_charges[no_charges_mask].copy()

            # Perform project_number fallback merge
            df_project_merged = pd.merge(
                df_no_charges.drop(columns=[col for col in charges_fields[1:] if col in df_no_charges.columns]),
                charges_project_unique,
                on="project_number",
                how="left"
            )

            # Update main dataframe with project_number results
            project_success_mask = ~df_project_merged['service_charges_total'].isna()
            update_indices = df_with_charges.index[no_charges_mask][project_success_mask]

            for col in charges_project_unique.columns:
                if col != "project_number":
                    df_with_charges.loc[update_indices, col] = df_project_merged.loc[project_success_mask, col].values

            project_success_count = project_success_mask.sum()
            logger.info(f"Project_number merge successful for {project_success_count} records")

        # Step 3: Fallback merge using prop-ai_location_key_building
        no_charges_mask = df_with_charges['service_charges_total'].isna()

        if no_charges_mask.sum() > 0 and "prop-ai_location_key_building" in charges_df.columns:
            logger.info(f"Step 3: Attempting prop-ai_location_key_building fallback merge for {no_charges_mask.sum()} records")

            # Prepare location key fallback merge  
            location_fallback_fields = [
                "prop-ai_location_key_building", "service_charge_group_name_en", "service_charges_budget_year",
                "service_charge_group_usage_en", "service_charges_general_fund", "service_charges_total"
            ]
            available_location_fields = [field for field in location_fallback_fields if field in charges_df.columns]

            # Create location key charges dataset
            charges_location = charges_df[available_location_fields].copy()

            # charges_location = charges_location.copy()

            # Remove duplicates - if multiple matches exist, ignore them
            charges_location = charges_location[~(charges_location["prop-ai_location_key_building"] == '')]
            charges_location = charges_location[~charges_location.isna()]
            charges_location_unique = charges_location.drop_duplicates(subset=["prop-ai_location_key_building"])

            logger.info(f"Removed {duplicate_mask.sum()} duplicate prop-ai_location_key_building records")

            # Get records that still need fallback merge
            df_still_no_charges = df_with_charges[no_charges_mask].copy()

            # Perform location key fallback merge
            df_location_merged = pd.merge(
                df_still_no_charges.drop(columns=[col for col in charges_fields[1:] if col in df_still_no_charges.columns]),
                charges_location_unique,
                on="prop-ai_location_key_building",
                how="left"
            )

            # Update main dataframe with location key results
            location_success_mask = ~df_location_merged['service_charges_total'].isna()
            update_indices = df_with_charges.index[no_charges_mask][location_success_mask]

            for col in charges_location_unique.columns:
                if col != "prop-ai_location_key_building":
                    df_with_charges.loc[update_indices, col] = df_location_merged.loc[location_success_mask, col].values

            location_success_count = location_success_mask.sum()
            logger.info(f"Location key merge successful for {location_success_count} records")

        # Final summary
        final_no_charges = df_with_charges['service_charges_total'].isna().sum()
        total_records = len(df_with_charges)
        logger.info(f"Final merge summary: {total_records - final_no_charges}/{total_records} records have service charges")

        return df_with_charges

    transactions_final = merge_properties_and_charges(transactions_df)
    mortgages_final = merge_properties_and_charges(mortgages_df)

    logger.info("Additional data merged successfully")
    return transactions_final, mortgages_final

    def process_mortgage_data(transactions_df, mortgages_df):
    """
    Match and enrich transactions with mortgage information.

    Matches mortgage records to transactions based primarily on property_id and instance_date,
    allowing a ±120-day window for matching. For transactions missing property_id, uses a fallback
    matching strategy based on a composite key of property attributes.

    Args:
        transactions_df (pd.DataFrame): DataFrame of property transactions.
        mortgages_df (pd.DataFrame): DataFrame of mortgage records.

    Returns:
        pd.DataFrame: transactions_df enriched with mortgage-related columns:
            - 'mortgage_transaction_id'
            - 'mortgage' (Yes/No)
            - 'mortgage_amount'
            - 'mortgage_percent' (mortgage_amount / actual_worth)
            - 'mortgage_type'
            - 'mortgage_type_id'

    Logs:
        Info about the start and successful completion of mortgage data processing.
    """
    logger.info("Processing mortgage data")

    # Convert dates
    transactions_df["instance_date"] = pd.to_datetime(transactions_df["instance_date"])
    mortgages_df["instance_date"] = pd.to_datetime(mortgages_df["instance_date"])

    # Filter mortgages to matching property IDs (for non-NA property_id cases)
    unique_property_ids = transactions_df["property_id"].dropna().unique()
    mortgages_filtered = mortgages_df[mortgages_df["property_id"].isin(unique_property_ids)]

    # Group mortgages by property_id
    mortgage_groups = mortgages_filtered.groupby("property_id")

    # For NA property_id cases, group mortgages by the mapping columns
    mapping_columns = ["property_type_prop-ai", "Number of Bedrooms", "prop-ai_location_key_building", "procedure_area"]

    # Create a composite key for mapping when property_id is NA
    def create_mapping_key(row):
        return tuple(row[col] for col in mapping_columns if col in row.index)

    # Group mortgages by mapping columns for NA property_id matching
    mortgages_na_mapping = {}
    for idx, mortgage in mortgages_df.iterrows():
        if pd.isna(mortgage.get("property_id")):
            key = create_mapping_key(mortgage)
            if key not in mortgages_na_mapping:
                mortgages_na_mapping[key] = []
            mortgages_na_mapping[key].append(mortgage)

    results = []
    for idx, transaction in tqdm(transactions_df.iterrows(), desc="Processing mortgage matches"):
        property_id = transaction["property_id"]
        transaction_id = transaction["transaction_id"]
        actual_worth = transaction.get("actual_worth", None)

        txn_date = transaction["instance_date"]

        mortgage_status = "No"
        mortgage_amount = mortgage_percent = mortgage_type = mortgage_type_id = None

        mortgage_transaction_id = None

        # Check if property_id is available and not NA
        if pd.notna(property_id) and property_id in mortgage_groups.groups:
            related_mortgages = mortgage_groups.get_group(property_id)

            # Filter by ±120 days
            date_range = (
                (related_mortgages["instance_date"] >= txn_date - timedelta(days=120)) & 
                (related_mortgages["instance_date"] <= txn_date + timedelta(days=120))
            )
            filtered = related_mortgages[date_range]

            if not filtered.empty:
                # Find closest match
                filtered = filtered.assign(date_diff=(filtered["instance_date"] - txn_date).abs())
                best_match = filtered.sort_values("date_diff").iloc[0]

                mortgage_transaction_id = best_match.get("transaction_id", None)
                mortgage_status = "Yes"
                mortgage_amount = best_match.get("actual_worth", 0)
                mortgage_percent = mortgage_amount / actual_worth if actual_worth else 0
                mortgage_type = "Mortgage Registration"
                mortgage_type_id = 13

        # If property_id is NA, try mapping on the specified columns
        elif pd.isna(property_id):
            mapping_key = create_mapping_key(transaction)

            if mapping_key in mortgages_na_mapping:
                related_mortgages_list = mortgages_na_mapping[mapping_key]
                related_mortgages = pd.DataFrame(related_mortgages_list)

                # Filter by ±120 days
                date_range = (
                    (related_mortgages["instance_date"] >= txn_date - timedelta(days=120)) & 
                    (related_mortgages["instance_date"] <= txn_date + timedelta(days=120))
                )
                filtered = related_mortgages[date_range]

                if not filtered.empty:
                    # Find closest match
                    filtered = filtered.assign(date_diff=(filtered["instance_date"] - txn_date).abs())
                    best_match = filtered.sort_values("date_diff").iloc[0]

                    mortgage_transaction_id = best_match.get("transaction_id", None)
                    mortgage_status = "Yes"
                    mortgage_amount = best_match.get("actual_worth", 0)
                    mortgage_percent = mortgage_amount / actual_worth if actual_worth else 0
                    mortgage_type = "Mortgage Registration"
                    mortgage_type_id = 13



        results.append({
            "transaction_id": transaction_id,
            "property_id": property_id,
            "mortgage_transaction_id": mortgage_transaction_id,
            "mortgage": mortgage_status,
            "mortgage_amount": mortgage_amount,
            "mortgage_percent": mortgage_percent,
            "mortgage_type": mortgage_type,
            "mortgage_type_id": mortgage_type_id
        })

    mortgage_results_df = pd.DataFrame(results)

    # Merge mortgage results back to transactions
    transactions_df = pd.merge(
        transactions_df, mortgage_results_df, 
        left_on="_id", right_on="transaction_id", 
        how="left", suffixes=('', '_mortgage')
    )

    # Clean up duplicate columns
    transactions_df.drop(['property_id_mortgage', 'transaction_id_mortgage', '_id'], axis=1, inplace=True, errors='ignore')

    logger.info("Mortgage data processing completed")
    return transactions_df

    def save_to_mongodb(df, collection, chunk_size=10000):
    """
    Save a DataFrame to a MongoDB collection using a temporary collection for atomic replacement.

    Args:
        df (pd.DataFrame): DataFrame to save.
        collection (pymongo.collection.Collection): Target MongoDB collection.
        chunk_size (int): Number of records per insert chunk.

    Raises:
        Exception: Propagates any insertion or rename errors.
    """
    logger.info(f"Saving {len(df)} records to {collection.name}")

    # Get database and create temporary collection name
    db = collection.database
    temp_collection_name = f"{collection.name}_temp"
    temp_collection = db[temp_collection_name]

    df = df.drop(columns=["_id"], errors='ignore')

    def dataframe_to_chunks(df, chunk_size):
        total_rows = len(df)
        for i in range(0, total_rows, chunk_size):
            chunk_df = df.iloc[i:i + chunk_size]
            chunk_records = []

            for record in chunk_df.to_dict('records'):
                for key, value in record.items():
                    if not isinstance(value, (list, dict)) and pd.isnull(value):
                        record[key] = None
                chunk_records.append(record)

            yield chunk_records

    try:
        # Step 1: Drop temporary collection if it exists
        temp_collection.drop()
        logger.info(f"Dropped existing temporary collection: {temp_collection_name}")

        # Step 2: Insert data into temporary collection
        for i, chunk in enumerate(dataframe_to_chunks(df, chunk_size), start=1):
            try:
                temp_collection.insert_many(chunk, ordered=False)
                logger.info(f"✅ Inserted chunk {i} ({len(chunk)} records) to temp collection")
            except Exception as e:
                logger.error(f"❌ Error inserting chunk {i} to temp collection: {e}")
                raise

        # Step 3: Rename collections (atomic operation)
        # First rename original collection to backup (if it exists)
        original_exists = collection.name in db.list_collection_names()
        backup_name = f"{collection.name}_backup"

        if original_exists:
            db[collection.name].rename(backup_name)
            logger.info(f"Renamed {collection.name} to {backup_name}")

        # Rename temp collection to target name
        temp_collection.rename(collection.name)
        logger.info(f"Renamed {temp_collection_name} to {collection.name}")

        # Step 4: Drop backup collection
        if original_exists:
            db[backup_name].drop()
            logger.info(f"Dropped backup collection: {backup_name}")

        logger.info(f"✅ Successfully saved {len(df)} records to {collection.name}")

    except Exception as e:
        logger.error(f"❌ Error during save operation: {e}")
        # Cleanup: try to restore from backup if rename was successful
        try:
            if original_exists and backup_name in db.list_collection_names():
                db[backup_name].rename(collection.name)
                logger.info(f"Restored {collection.name} from backup")
        except Exception as restore_error:
            logger.error(f"❌ Error restoring from backup: {restore_error}")

        # Drop temp collection if it still exists
        try:
            if temp_collection_name in db.list_collection_names():
                temp_collection.drop()
                logger.info(f"Cleaned up temporary collection: {temp_collection_name}")
        except Exception as cleanup_error:
            logger.error(f"❌ Error cleaning up temp collection: {cleanup_error}")

        raise
