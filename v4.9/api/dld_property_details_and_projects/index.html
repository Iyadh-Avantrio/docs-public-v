    Property Details and Projects

    This module orchestrates the full ETL workflow for Dubai Land Department (DLD) data, including projects and property details.

    Key Responsibilities¶

    • Download and upload raw DLD source CSV files to S3.
    • Process and clean projects data; merge with transactions and location mappings.
    • Process property details (buildings, units, lands) with source tagging, cleaning, project merges, and location enrichment.
    • Save cleaned and merged data to MongoDB collections, with archiving and atomic collection swaps.
    • Provide utility functions for data loading, cleaning, merging, and location mapping.
    • Manage MongoDB collections with safe rename and delete operations.
    • Ensure memory cleanup and logging throughout processing.

    Outputs¶

    • Cleaned and enriched projects and property details saved to MongoDB.
    • Archived previous project records on updates.
    • Combined property details collection for comprehensive property data access.

    This pipeline enables reliable, consistent, and enriched DLD data integration for downstream analytics and applications.

    def clean_projects_dataframe(df):
    """
    Clean the DLD projects dataframe by:
    - Dropping unnecessary Arabic language and description columns.
    - Converting ID columns to string and cleaning trailing decimals.
    - Uppercasing and stripping key text columns.
    - Parsing date columns to datetime objects.

    Args:
        df (pd.DataFrame): Raw projects dataframe.

    Returns:
        pd.DataFrame: Cleaned projects dataframe.
    """
    logging.info('Cleaning projects dataframe')

    # Drop unnecessary columns
    columns_to_drop = [
        'project_name', 'developer_name', 'master_developer_name',
        'project_type_ar', 'project_classification_ar', 'escrow_agent_name',
        'project_status_ar', 'project_description_ar', 'area_name_ar',
        'master_project_ar', 'zoning_authority_ar'
    ]

    df = df.drop([col for col in columns_to_drop if col in df.columns], axis=1)

    # Convert ID columns to string and clean
    id_columns = [
        'project_id', 'project_number', 'developer_id', 'developer_number',
        'master_developer_id', 'master_developer_number', 'project_type_id',
        'project_classification_id', 'escrow_agent_id', 'property_id',
        'area_id', 'zoning_authority_id'
    ]

    for col in id_columns:
        if col in df.columns:
            df[col] = df[col].astype(str)
            if col in ['developer_id', 'developer_number', 'escrow_agent_id', 'area_id']:
                df[col] = df[col].str.rstrip('.0')

    # Clean text columns
    columns_to_process = ['area_name_en', 'project_name_en', 'master_project_en']
    for col in columns_to_process:
        if col in df.columns and df[col].dtype == 'object':
            df[col] = df[col].str.upper().str.strip()

    # Convert date columns
    date_columns = ['project_start_date', 'project_end_date', 'completion_date', 'cancellation_date']
    for col in date_columns:
        if col in df.columns:
            df[col] = pd.to_datetime(df[col], dayfirst=True, format='mixed')

    return df

    def create_dataframe_from_db_dld_location_mapping():

    def dld_projects_update(db):
    """
    Process and update DLD projects data from S3 CSV to MongoDB.

    This function:
    - Loads a compressed DLD projects CSV from S3.
    - Cleans and standardizes the DataFrame.
    - Merges it with unique project numbers from transactions for validation.
    - Maps location details from a location mapping collection.
    - Saves updated projects to MongoDB with archiving for changed records.
    - Frees memory after processing.

    Args:
        db (pymongo.database.Database): The MongoDB database instance where project data will be saved.

    Raises:
        Exception: If any processing or saving step fails.
    """
    logging.info('Starting DLD projects update')

    try:
        # Read projects data
        dld_projects_df = pd.read_csv(
            f's3://{TRANSACTION_BUCKET}/source_data/dld_data/{current_date}_dld_projects.csv',
            header=0, 
            compression='gzip', 
            storage_options={"key": ACCESS_ID, "secret": ACCESS_KEY}
        )

        logging.info(f"Loaded {len(dld_projects_df)} project records")

        # Clean dataframe
        cleaned_dld_projects_df = clean_projects_dataframe(dld_projects_df)
        logging.info(f"After cleaning: {len(cleaned_dld_projects_df)} project records")

        # Get unique project numbers from transactions
        dld_transaction_df = create_dataframe_from_db_dld_transactions()
        logging.info(f"Found {len(dld_transaction_df)} unique projects in transactions")

        cleaned_dld_projects_df['project_number'] = pd.to_numeric(cleaned_dld_projects_df['project_number'], errors='coerce').astype('Int64')
        dld_transaction_df['project_number'] = pd.to_numeric(dld_transaction_df['project_number'], errors='coerce').astype('Int64')

        # Merge with transactions
        cleaned_dld_projects_df = cleaned_dld_projects_df.merge(
            dld_transaction_df, on='project_number', how='left'
        )

        # Get location mapping
        dld_location_mapping_df = create_dataframe_from_db_dld_location_mapping()
        cleaned_dld_projects_df = map_location_data(cleaned_dld_projects_df, dld_location_mapping_df)

        # Process and save to database
        save_projects_to_db(db, cleaned_dld_projects_df)

        logging.info('DLD projects update completed')

    except Exception as e:
        logging.error(f"Error in DLD projects update: {str(e)}")
        raise
    finally:
        # Cleanup
        del dld_projects_df, cleaned_dld_projects_df
        gc.collect()

    def dld_property_details_update(db):
    """
    Process and update DLD property details for buildings, units, and lands.

    This function:
    - Reads compressed CSVs from S3 for each property type.
    - Adds source identifier and converts ID fields to numeric format.
    - Cleans and standardizes relevant text columns.
    - Merges with existing projects (if available).
    - Maps location metadata using project and location mapping.
    - Saves individual and combined property details into MongoDB collections.
    - Performs memory cleanup after execution.

    Args:
        db (pymongo.database.Database): The MongoDB database instance where property details will be saved.

    Raises:
        Exception: If any part of the update process fails.
    """
    logging.info('Starting DLD property details update')

    try:
        # Read and clean property data
        property_data = {}
        property_types = ['buildings', 'units', 'lands']

        # Load project data
        projects_df = get_projects_for_merge(db)
        if not projects_df.empty:
            projects_df['project_id'] = pd.to_numeric(projects_df['project_id'], errors='coerce').astype('Int64')
        else:
            logging.warning("Project data is empty. Skipping project merge.")
            projects_df = None


        for prop_type in property_types:
            df = pd.read_csv(
                f's3://{TRANSACTION_BUCKET}/source_data/dld_data/{current_date}_dld_{prop_type}.csv',
                header=0, compression='gzip',
                storage_options={"key": ACCESS_ID, "secret": ACCESS_KEY}
            )
            df['Source_Data'] = f'dld_{prop_type}'


            # Convert property_id and project_id to numeric
            df['property_id'] = pd.to_numeric(df['property_id'], errors='coerce').astype('Int64')
            df['project_id'] = pd.to_numeric(df['project_id'], errors='coerce').astype('Int64')

            # Clean text columns
            columns_to_process = ['area_name_en', 'project_name_en', 'master_project_en']
            for col in columns_to_process:
                if col in df.columns and df[col].dtype == 'object':
                    df[col] = df[col].str.upper().str.strip()

            # Merge with projects_df if available
            if projects_df is not None:
                duplicate_cols = [col for col in projects_df.columns if col in df.columns and col != 'project_id']
                projects_df_reduced = projects_df.drop(columns=duplicate_cols)
                df = df.merge(projects_df_reduced, on='project_id', how='left')
                logging.info(f"Merged {prop_type} with projects, resulting in {len(df)} records")

            dld_location_mapping_df = create_dataframe_from_db_dld_location_mapping()

            df = map_location_data(df, dld_location_mapping_df)

            property_data[prop_type] = df

            # Save individual property collection (merged)
            save_property_collection(db, df, f'dld_{prop_type}')

        # Combine all merged property data
        combined_df = pd.concat(list(property_data.values()), ignore_index=True)
        logging.info(f"Combined total of {len(combined_df)} property records")

        # Save final merged collection
        save_property_collection(db, combined_df, 'dld_property_details')

        logging.info('DLD property details update completed')

    except Exception as e:
        logging.error(f"Error in DLD property details update: {str(e)}")
        raise
    finally:
        # Cleanup
        for df in property_data.values():
            del df
        del combined_df
        gc.collect()

    def map_location_data(projects_df, location_df):
    """
    Map location data from location_df to projects_df by:
    - Primary join on 'project_number'.
    - Fallback join on composite keys ['area_name_en', 'master_project_en', 'project_name_en'] for missing locations.

    Adds empty columns 'building_name_en' and 'building_name_prop-ai' to final DataFrame.

    Args:
        projects_df (pd.DataFrame): Projects DataFrame.
        location_df (pd.DataFrame): Location mapping DataFrame.

    Returns:
        pd.DataFrame: Projects DataFrame enriched with location mapping.
    """

    if location_df.empty:
        logging.warning("Location mapping DataFrame is empty")
        return projects_df

    logging.info('Applying location mapping')

    # Ensure proper types
    projects_df['project_number'] = pd.to_numeric(projects_df['project_number'], errors='coerce').astype('Int64')
    location_df['project_number'] = pd.to_numeric(location_df['project_number'], errors='coerce').astype('Int64')

    # Drop duplicates in location_df based on project_number
    location_by_number = location_df[location_df['project_number'].notna()].drop_duplicates(subset=['project_number'])

    # Identify common columns (excluding project_number)
    common_cols = list(set(projects_df.columns) & set(location_by_number.columns)) 
    common_cols.remove('project_number')

    # Drop common columns from left (projects_df)
    projects_df = projects_df.drop(columns=common_cols, errors='ignore')

    # First merge on project_number
    merged_df = projects_df.merge(location_by_number, on='project_number', how='left')

    # SECONDARY MERGE (fallback) using composite key
    composite_keys = ['area_name_en', 'master_project_en', 'project_name_en']
    missing_locations = merged_df[merged_df['project_number'].isna()]

    # Drop common columns from left before second merge
    missing_locations = missing_locations.drop(columns=[col for col in missing_locations.columns if col in location_df.columns and col not in composite_keys], errors='ignore')
    location_by_keys = location_df.dropna(subset=composite_keys).drop_duplicates(subset=composite_keys)

    # Merge on composite keys
    merged_fallback = missing_locations.merge(location_by_keys, on=composite_keys, how='left')

    # Combine the successfully merged rows with fallback rows
    successfully_mapped = merged_df[~merged_df['area_name_en'].isna()]
    final_df = pd.concat([successfully_mapped, merged_fallback], ignore_index=True)

    final_df["building_name_en"] = None
    final_df["building_name_prop-ai"] = None

    logging.info('Location mapping completed with project_number and composite key fallback')
    return final_df

    def merge_with_projects(property_df, projects_df):
    """
    Merge property DataFrame with projects DataFrame on 'project_id'.

    Drops columns from projects_df that would duplicate property_df columns (except 'project_id').

    Args:
        property_df (pd.DataFrame): Property details DataFrame.
        projects_df (pd.DataFrame): Projects DataFrame.

    Returns:
        pd.DataFrame: Merged DataFrame.
    """
    logging.info('Merging property data with projects')

    # Convert project_id to integer for merging
    property_df['project_id'] = pd.to_numeric(property_df['project_id'], errors='coerce').astype('Int64')
    property_df = property_df.dropna(subset=['project_id'])

    common_columns = set(property_df.columns) & set(projects_df.columns)

    projects_df_clean = projects_df.drop(columns=list(common_columns))

    # Merge
    merged_df = property_df.merge(projects_df_clean, on='project_id', how='left')

    logging.info(f'Merged data contains {len(merged_df)} records')
    return merged_df

    def prepare_source_data():
    """
    Download source data files from predefined URLs and upload them to S3.

    Downloads CSV files for projects, units, lands, and buildings, then uploads
    each to the designated S3 bucket with the current date as prefix.

    Logs progress and errors.
    """
    logging.info('Starting source data preparation')

    data_sources = [
        (DLD_PROJECTS_URL, f'{current_date}_dld_projects.csv'),
        (DLD_UNITS_URL, f'{current_date}_dld_units.csv'),
        (DLD_LANDS_URL, f'{current_date}_dld_lands.csv'),
        (DLD_BUILDINGS_URL, f'{current_date}_dld_buildings.csv')
    ]

    for url, filename in data_sources:
        try:
            response = requests.get(url, stream=True)

            if response.status_code == 200:
                s3_object_key = f'{DLD_DATA_INPUT_BUCKET_PREFIX}/{filename}'
                s3_client.upload_fileobj(response.raw, TRANSACTION_BUCKET, s3_object_key)

        except Exception as e:
            logging.error(f"Error processing {filename}: {str(e)}")

    logging.info('Source data preparation completed')

    def rename_collection(old_name, new_name, db):
    """Rename a MongoDB collection"""
    try:
        if old_name not in db.list_collection_names():
            logging.error(f"Collection '{old_name}' does not exist")
            return

        if new_name in db.list_collection_names():
            logging.error(f"Collection '{new_name}' already exists")

    def save_projects_to_db(db, df):
    """
    Save the cleaned projects dataframe to MongoDB, maintaining an archive for updated records.

    For each project record:
    - Checks if a record with same project_number exists.
    - If exists and data changed, archive old record with timestamps.
    - Insert or update current record with updated timestamp.

    Args:
        db (pymongo.database.Database): MongoDB database handle.
        df (pd.DataFrame): Cleaned projects dataframe.
    """
    logging.info('Saving projects to database')

    primary_key = 'project_number'
    collection_name = 'dld_projects'
    archive_collection_name = 'dld_projects_archive'

    # Clear archive collection
    db[archive_collection_name].delete_many({})

    records = df.to_dict(orient='records')

    # Process date fields
    for record in records:
        for date_field in ['project_start_date', 'project_end_date', 'completion_date', 'cancellation_date']:
            if record.get(date_field):
                try:
                    if isinstance(record[date_field], str):
                        record[date_field] = datetime.strptime(record[date_field], '%Y-%m-%dT%H:%M:%S.%f%z')
                    elif pd.isna(record[date_field]):
                        record[date_field] = None
                except (ValueError, TypeError):
                    record[date_field] = None

    # Update with archiving logic
    for record in records:
        existing_document = db[collection_name].find_one({primary_key: record[primary_key]})
        if existing_document:
            # Check for updates
            has_updates = any(
                record[key] != existing_document.get(key)
                for key in record if key != 'property_id'
            )
            if has_updates:
                # Archive existing document
                existing_document['date_updated'] = datetime.now()
                existing_document.pop('_id', None)
                existing_document['archived_at'] = datetime.now()
                db[archive_collection_name].insert_one(existing_document)

                # Replace with new document
                record['date_updated'] = datetime.now()
                db[collection_name].replace_one({primary_key: record[primary_key]}, record, upsert=True)
        else:
            # Insert new record
            record['date_updated'] = datetime.now()
            db[collection_name].insert_one(record)

    logging.info(f'Saved {len(records)} projects to database')

    def save_property_collection(db, df, collection_name):
    """
    Save property data to a MongoDB collection using a temporary collection swap approach.

    - Converts NaNs to None for MongoDB compatibility.
    - Clears a temporary collection and inserts all records.
    - Drops the original collection and renames the temp collection atomically.

    Args:
        db (pymongo.database.Database): MongoDB database object.
        df (pd.DataFrame): Property data to save.
        collection_name (str): Target MongoDB collection name.
    """
    logging.info(f'Saving {len(df)} records to {collection_name}')

    temp_collection_name = f'{collection_name}_temp'
    records = df.drop(columns=['_id'], errors='ignore').to_dict(orient='records')

    # Convert NaNs to None
    for record in records:
        for key, value in record.items():
            if not isinstance(value, (list, dict)) and pd.isnull(value):
                record[key] = None

    if records:
        # Drop temp collection if it exists
        db[temp_collection_name].delete_many({})  # Clear it instead of drop to avoid index loss if any

        # Insert all records
        db[temp_collection_name].insert_many(records)

        # Replace old collection
        db[collection_name].drop()
        db[temp_collection_name].rename(collection_name)

    logging.info(f'Successfully saved to {collection_name}')
