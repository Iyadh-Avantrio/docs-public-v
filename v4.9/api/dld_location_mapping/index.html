    Location Mapping

    This module handles the standardization and enrichment of location data for Dubai Land Department (DLD) listings by applying a multi-step mapping process. It combines raw property and transaction data with curated mapping files and MongoDB records to produce clean, consistent location labels in the prop-ai naming format.

    Key Responsibilities¶

    • Load DLD transactions, rent contracts, and multiple mapping files from S3.
    • Clean and preprocess raw location fields from DLD source data.

    • Project number
    • Exact multi-field matches
    • Single field fallback matches
    • Enrich records with hierarchical labels: building, project, master project, area.
    • Generate structured mapping keys and classification flags.
    • Log anomalies and missing mappings (e.g. unmatched master_project_en).
    • Persist results to both CSV and MongoDB collections.

    Outputs¶

    • A fully mapped and classified DataFrame with enriched prop-ai fields.
    • Optional hierarchical records to supplement project and area hierarchy.
    • Saved outputs in both .csv format and MongoDB collections.

    This pipeline ensures consistent and hierarchical geolocation data across DLD sources, which is crucial for search, filtering, reporting, and property aggregation logic.

    def clean_text_columns(df, columns, placeholder=""):
    """
    Clean and standardize specified text columns in a DataFrame.

    This function uppercases, trims leading/trailing whitespace, replaces multiple spaces with a 
    single space, and replaces instances of `' , '` with `', '` for all specified text columns. 
    It also replaces `NaN` and empty strings with a placeholder value.

    Args:
        df (pd.DataFrame): The input DataFrame.
        columns (list[str]): List of column names to clean.
        placeholder (str, optional): Value to replace `NaN` and empty strings. Defaults to `""`.

    Returns:
        pd.DataFrame: The cleaned DataFrame with standardized text columns.
    """
    for col in columns:
        if df[col].dtype == "object":
            df[col] = df[col].str.upper().str.strip().str.replace(" +", " ", regex=True).str.replace(" , ", ", ", regex=False).fillna(placeholder).replace("", placeholder)
    return df

    def clean_value(val):
    """
    Clean string values by handling various null representations.

    Args:
        val (any): Value to clean.

    Returns:
        any or None: Cleaned value or None if considered null.
    """
    if pd.isna(val):
        return val
    if isinstance(val, str):
        return val.strip() if val.strip() not in ["None", "NONE", "", "null", "nan"] else None
    return val if val not in [None, "None", "NONE", "", "null", np.nan, 'nan'] else None

    def determine_mapping_type(row):
    """
    Determine the mapping type for a row based on available fields.

    Args:
        row (pd.Series): DataFrame row.

    Returns:
        str: One of ['building', 'project', 'master_project', 'area']
    """
    if pd.notna(row['building_name_en']):
        return "building"
    elif pd.notna(row['project_name_en']):
        return "project"
    elif pd.notna(row['master_project_en']):
        return "master_project"
    else:
        return "area"

    def enrich_with_labels(locations_df):
    """
    Enrich the locations DataFrame with human-readable labels by merging data from MongoDB collections.

    Args:
        locations_df (pd.DataFrame): DataFrame with location keys to enrich.

    Returns:
        pd.DataFrame: Enriched DataFrame with label columns merged.
    """
        # Connect to databases
    dev_client = MongoClient(os.getenv('DEV_MONGO_DB_URL'))

    dev_db = dev_client['dev_listings_db']

    # Load reference data
    projects_df = pd.DataFrame(list(dev_db['data_projects'].find()))
    master_projects_df = pd.DataFrame(list(dev_db['data_master_projects'].find()))
    areas_df = pd.DataFrame(list(dev_db['data_areas'].find()))

    # Create keys in projects_df
    projects_df['prop-ai_location_key_building'] = make_key(projects_df, [
        'area_name_prop-ai', 'master_project_prop-ai', 'project_name_prop-ai', 'building_name_prop-ai'
    ])

    projects_df['prop-ai_location_key_project'] = make_key(projects_df, [
        'area_name_prop-ai', 'master_project_prop-ai', 'project_name_prop-ai'
    ])

    # Merge building_name_label
    if 'building_name_label' in projects_df.columns:
        building_labels = projects_df[['prop-ai_location_key_building', 'building_name_label']].drop_duplicates()
        locations_df = locations_df.merge(building_labels, on='prop-ai_location_key_building', how='left')

    # Merge project_name_label
    if 'project_name_label' in projects_df.columns:
        project_labels = projects_df[['prop-ai_location_key_project', 'project_name_label']].drop_duplicates()
        locations_df = locations_df.merge(project_labels, on='prop-ai_location_key_project', how='left')

    # Create key in master_projects_df and merge master_project_label
    master_projects_df['prop-ai_location_key_masterproject'] = make_key(master_projects_df, [
        'area_name_prop-ai', 'master_project_prop-ai'
    ])

    if 'master_project_label' in master_projects_df.columns:
        master_labels = master_projects_df[['prop-ai_location_key_masterproject', 'master_project_label']].drop_duplicates()
        locations_df = locations_df.merge(master_labels, on='prop-ai_location_key_masterproject', how='left')

    # Create key in areas_df and merge area_name_label
    areas_df['prop-ai_location_key_area'] = areas_df['area_name_prop-ai'].fillna('')

    if 'area_name_label' in areas_df.columns:
        area_labels = areas_df[['prop-ai_location_key_area', 'area_name_label']].drop_duplicates()
        locations_df = locations_df.merge(area_labels, on='prop-ai_location_key_area', how='left')

    # Remove duplicate columns that may have been created during merging
    locations_df = locations_df.loc[:, ~locations_df.columns.str.endswith(('_x', '_y'))]

    return locations_df

    def generate_hierarchical_records(df_combined):
    """
    Generate additional hierarchical records for projects, master projects, and areas
    based on existing mapped records.

    Args:
        df_combined (pd.DataFrame): DataFrame with mapping data.

    Returns:
        list[dict]: List of new records to append.
    """
    new_records = []

    def record_exists(df, mapping_type, area, master, project):
        return (
            (df['mapping_type'] == mapping_type) &
            (df['area_name_en'] == area) &
            (df['master_project_en'] == master) &
            (df['project_name_en'] == project)
        ).any()

    for _, row in df_combined.iterrows(): 
        # if pd.notna(row["project_number"]): 
        #     continue

        area = row['area_name_en']
        master = row['master_project_en']
        project = row['project_name_en']

        base_fields = {
            'area_name_en': area,
            'area_name_prop-ai': row['area_name_prop-ai'],
            'building_name_en': None,
            'building_name_prop-ai': None,
            'project_name_en': None,
            'project_name_prop-ai': None,
            'master_project_en': None,
            'master_project_prop-ai': None,
            'project_number': None,
            'Source': row['Source'],
            'mapped_status': True,
            'auto_update': True,
            'updated_at': pd.Timestamp.now()
        }

        if row['mapping_type'] == 'building':
            # Add project record if doesn't exist
            if not record_exists(df_combined, 'project', area, master, project):
                new_records.append({
                    **base_fields,
                    'mapping_type': 'project',
                    'project_name_en': project,
                    'project_name_prop-ai': row['project_name_prop-ai'],
                    'master_project_en': master,
                    'master_project_prop-ai': row['master_project_prop-ai'],
                })

            # Add master_project record if doesn't exist
            if not record_exists(df_combined, 'master_project', area, master, None):
                new_records.append({
                    **base_fields,
                    'mapping_type': 'master_project',
                    'master_project_en': master,
                    'master_project_prop-ai': row['master_project_prop-ai'],
                })

            # Add area record if doesn't exist
            if not record_exists(df_combined, 'area', area, None, None):
                new_records.append({
                    **base_fields,
                    'mapping_type': 'area'
                })

        elif row['mapping_type'] == 'project':
            # Add master_project record if doesn't exist
            if not record_exists(df_combined, 'master_project', area, master, None):
                new_records.append({
                    **base_fields,
                    'mapping_type': 'master_project',
                    'master_project_en': master,
                    'master_project_prop-ai': row['master_project_prop-ai'],
                })

            # Add area record if doesn't exist
            if not record_exists(df_combined, 'area', area, None, None):
                new_records.append({
                    **base_fields,
                    'mapping_type': 'area'
                })

        elif row['mapping_type'] == 'master_project':
            # Add area record if doesn't exist
            if not record_exists(df_combined, 'area', area, None, None):
                new_records.append({
                    **base_fields,
                    'mapping_type': 'area'
                })

    return new_records

    def generate_location_keys(df_combined):
    """
    Generate concatenated location keys based on prop-ai naming conventions.

    Args:
        df_combined (pd.DataFrame): DataFrame containing prop-ai location fields.

    Returns:
        pd.DataFrame: DataFrame with new location key columns added.
    """
    df_combined['prop-ai_location_key'] = (
        df_combined['area_name_prop-ai'].fillna('') + "_" + 
        df_combined['master_project_prop-ai'].fillna('') + "_" + 
        df_combined['project_name_prop-ai'].fillna('')
    )

    df_combined['prop-ai_location_key_building'] = (
        df_combined['area_name_prop-ai'].fillna('') + "_" + 
        df_combined['master_project_prop-ai'].fillna('') + "_" + 
        df_combined['project_name_prop-ai'].fillna('') + "_" + 
        df_combined['building_name_prop-ai'].fillna('')
    )

    df_combined['prop-ai_location_key_project'] = (
        df_combined['area_name_prop-ai'].fillna('') + "_" + 
        df_combined['master_project_prop-ai'].fillna('') + "_" + 
        df_combined['project_name_prop-ai'].fillna('')
    )

    df_combined['prop-ai_location_key_masterproject'] = (
        df_combined['area_name_prop-ai'].fillna('') + "_" + 
        df_combined['master_project_prop-ai'].fillna('')
    )

    df_combined['prop-ai_location_key_area'] = df_combined['area_name_prop-ai'].fillna('')

    return df_combined

        'building_name_prop-ai'
    ]

    for col in prop_ai_columns:
        if col not in df_combined.columns:
            df_combined[col] = None

    return df_combined

    def load_dld_data_from_s3():
    """
    Load DLD transaction and rental contract data from S3, decompress and return as DataFrames.

    Returns:
        tuple:
            pandas.DataFrame: DLD transactions data.
            pandas.DataFrame: DLD rent contracts data.

    Raises:
        Exception: If loading from S3 or decompression fails.
    """
    s3_client = get_s3_client()
    today = datetime.now().strftime('%Y%m%d')

    # Define S3 paths
    transactions_key = f'source_data/dld_data/{today}_dld_transactions.csv'
    rent_contracts_key = f'source_data/dld_data/{today}_dld_rent_contracts.csv'
    bucket_name = 'prop-ai-transactions'

    # Define local file paths
    local_transactions_file = f'temp_{today}_dld_transactions.csv.gz'
    local_rent_contracts_file = f'temp_{today}_dld_rent_contracts.csv.gz'

    # Load transactions
    logger.info(f"Loading DLD transactions from S3: s3://{bucket_name}/{transactions_key}")
    try:
        # Download to local file
        s3_client.download_file(bucket_name, transactions_key, local_transactions_file)

        # Load from local file
        with gzip.open(local_transactions_file, 'rt') as gz:
            df_transactions = pd.read_csv(gz)
        logger.info(f"Successfully loaded {len(df_transactions)} transaction records")
    except Exception as e:
        logger.error(f"Failed to load transactions from S3: {str(e)}")
        raise

    # Load rent contracts
    logger.info(f"Loading DLD rent contracts from S3: s3://{bucket_name}/{rent_contracts_key}")
    try:
        # Download to local file
        s3_client.download_file(bucket_name, rent_contracts_key, local_rent_contracts_file)

        # Load from local file
        with gzip.open(local_rent_contracts_file, 'rt') as gz:
            df_rent_contracts = pd.read_csv(gz)
        logger.info(f"Successfully loaded {len(df_rent_contracts)} rent contract records")
    except Exception as e:
        logger.error(f"Failed to load rent contracts from S3: {str(e)}")
        raise

    # Clean up local files
    try:
        if os.path.exists(local_transactions_file):
            os.remove(local_transactions_file)
        if os.path.exists(local_rent_contracts_file):
            os.remove(local_rent_contracts_file)
        logger.info("Local temporary files cleaned up")
    except Exception as e:
        logger.warning(f"Failed to clean up temporary files: {str(e)}")

    # Filter to required fields
    required_fields = [
        'area_name_en', 'master_project_en', 'project_name_en', 'building_name_en', 'project_number',
        'area_name_prop-ai', 'master_project_prop-ai', 'project_name_prop-ai', 'building_name_prop-ai'
    ]

    # Keep only required fields that exist in the dataframes
    transactions_fields = [field for field in required_fields if field in df_transactions.columns]
    rent_contracts_fields = [field for field in required_fields if field in df_rent_contracts.columns]

    df_transactions = df_transactions[transactions_fields]
    df_rent_contracts = df_rent_contracts[rent_contracts_fields]

    df_rent_contracts["building_name_en"] = ""

    return df_transactions, df_rent_contracts

    unmapped_mask = (df_combined['mapped_status'].isna()) | (df_combined['mapped_status'] == False)
    unmapped_indices = df_combined[unmapped_mask].index

        for field in fields:

                # Case-insensitive comparison

    def save_results(df_combined, output_file="mapped_data_output.csv"):
    """
    Save the mapped DataFrame to CSV file and MongoDB collection.

    Args:
        df_combined (pd.DataFrame): The final DataFrame with mapping results.
        output_file (str): Filename for saving CSV output. Defaults to "mapped_data_output.csv".
    """
    try:
        # Save to CSV
        df_combined.to_csv(output_file, index=False)
        logger.info(f"Successfully saved mapped data to {output_file}")

        # Save to MongoDB
        dev_client = MongoClient(os.getenv('DEV_MONGO_DB_URL'))
        dev_db = dev_client['dev_listings_db']

        # Clear existing collection and insert new data
        dev_db['propai_dld_locations_mapping_v5'].delete_many({})
        dev_db['propai_dld_locations_mapping_v5'].insert_many(df_combined.to_dict(orient='records'))

        logger.info("Successfully saved data to MongoDB")

    except Exception as e:
        logger.error(f"Failed to save results: {str(e)}")
