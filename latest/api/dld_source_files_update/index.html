    Source Files Update

    This module handles ingestion, cleaning, merging, and uploading of DIFC transactional, property, and service charge data.

    Core Functionalities¶

    • Initialize AWS S3 client and upload local or downloaded files to S3.
    • Download source CSVs from Dubai Pulse or remote SFTP servers and upload to S3.
    • Clean and convert data columns (strings, dates, numerics) with type coercion.
    • Load and preprocess transactions, properties, and service charges datasets from S3 or local files.
    • Merge transactional and property datasets for consolidated views.
    • Upload large datasets to MongoDB in batches with logging and error handling.
    • Cleanup local downloaded files after processing.
    • Orchestrate full ETL workflow via a main entrypoint function.

    Usage¶

    Supports flexible data source configuration (S3 or local), batch size tuning for uploads, and robust error management.

    This pipeline ensures consistent, clean, and integrated DIFC data is available in MongoDB for downstream analytics and applications.

    def clean_and_convert_column(series, target_type):
    """
    Clean and convert a pandas Series column to a specified target type.

    Args:
        series (pandas.Series): The data column to clean and convert.
        target_type (str): The target data type. One of: 'string', 'date', 'float', 'int'.

    Returns:
        (pandas.Series): Converted and cleaned Series.

    Notes:
        - Strings "nan", "None", "null" are replaced with None for 'string' target.
        - Dates are parsed with errors coerced to NaT.
        - Numeric conversions coerce errors to NaN/nullable integers.
    """
    if target_type == "string":
        return series.astype(str).replace(["nan", "None", "null"], None)

    elif target_type == "date":
        return pd.to_datetime(series, errors="coerce")

    elif target_type == "float":
        return pd.to_numeric(series, errors="coerce").astype("float64")

    elif target_type == "int":
        return pd.to_numeric(series, errors="coerce").astype(
            "Int64"
        )  # Nullable integer

    return series

    def download_files_via_sftp():
    """
    Download specified CSV files from a remote server via SFTP and upload them to S3.

    Uses paramiko to connect to the remote SFTP server and download files locally,
    then uploads each file to an S3 bucket with the current date suffix.

    Raises:
        Exception: If any connection, download, or upload operation fails.
    """
    logger.info("Connecting to remote server via SFTP to download files...")
    today = current_date  # Use the global current_date variable

    try:
        ssh = paramiko.SSHClient()
        ssh.set_missing_host_key_policy(paramiko.AutoAddPolicy())
        ssh.connect(
            hostname=DLD_SFTP_HOST,
            username=DLD_SFTP_USER_NAME,
            password=DLD_SFTP_PASSWORD,
            timeout=30,
            compress=True,
        )

        sftp = ssh.open_sftp()

        try:
            for filename in CSV_FILES:
                remote_file = f"{DLD_SFTP_TRANSACTIONS_PATH}/{filename}"
                local_file = LOCAL_DATA_DIR / filename

                # Download file locally first
                logger.info(f"Downloading: {remote_file}")
                sftp.get(remote_file, str(local_file))
                logger.info(f"✅ Successfully downloaded: {filename}")

                # Upload to S3 with today's date suffix
                base_name = filename.replace(".csv", "")
                s3_filename = f"{base_name}_{today}.csv"
                s3_key = f"{S3_PREFIX}{s3_filename}"
                upload_to_s3(local_file, s3_key)

        finally:
            sftp.close()
            ssh.close()

        logger.info("✅ All files downloaded and uploaded to S3 successfully.")

    except Exception as e:
        logger.error(f"❌ Failed to download files: {str(e)}")
        raise

    def load_and_process_properties(use_s3=True):
    """
    Load and process properties data with enforced string dtypes for ID columns.

    Args:
        use_s3 (bool): Whether to load data from S3 or local files. Defaults to True.

    Returns:
        pandas.DataFrame: Processed properties data.
    """
    logger.info("Loading properties data...")

    if use_s3:
        # S3 configuration
        bucket_name = "prop-ai-transactions"
        key = "source_data/dld_data_properties/20250531_data_b2b_trans_properties_conf.csv"

        try:
            logger.info(f"Loading properties from S3: s3://{bucket_name}/{key}")
            response = s3_client.get_object(Bucket=bucket_name, Key=key)

            # Read CSV directly from S3 response
            df_raw = pd.read_csv(
                response['Body'], 
                dtype=PROPERTIES_DTYPES, 
                low_memory=False
            )
            logger.info(f"Loaded {len(df_raw)} property records from S3 with enforced string dtypes")

        except ClientError as e:
            error_code = e.response['Error']['Code']
            if error_code == 'NoSuchKey':
                raise FileNotFoundError(f"Properties file not found in S3: s3://{bucket_name}/{key}")
            elif error_code == 'NoSuchBucket':
                raise FileNotFoundError(f"S3 bucket not found: {bucket_name}")
            else:
                raise Exception(f"Error loading properties from S3: {e}")
        except Exception as e:
            raise Exception(f"Unexpected error loading properties from S3: {e}")
    else:
        # Fallback to local file loading
        file_path = LOCAL_DATA_DIR / CSV_FILES[1]

        if not file_path.exists():
            raise FileNotFoundError(f"Properties file not found: {file_path}")

        # Load with enforced string dtypes for ID columns
        df_raw = pd.read_csv(file_path, dtype=PROPERTIES_DTYPES, low_memory=False)
        logger.info(f"Loaded {len(df_raw)} property records with enforced string dtypes")

    df_processed = process_properties_data(df_raw.copy())

    return df_processed

    """

    This function downloads the DLD transactions and rent contracts CSV files,
    then uploads each to a specified S3 bucket with date-prefixed keys.

    try:
        # Upload transaction file
        response = requests.get(DLD_TRANSACTION_URL, stream=True)
        if response.status_code == 200:

                f"{DLD_DATA_INPUT_BUCKET_PREFIX}/{current_date}_dld_transactions.csv"

        response = requests.get(DLD_RENT_CONTRACTS_URL, stream=True)

    def process_properties_data(df):
    """
    Process properties data with appropriate type conversions.

    Args:
        df (pandas.DataFrame): Raw properties data.

    Returns:
        pandas.DataFrame: Processed properties data with cleaned and converted columns.

    logger.info("Processing properties data...")

    # Convert column names to lowercase

        print(f"Column: {col}, Dtype: {df[col].dtype}")

    # Apply data type conversions - ID columns are already strings from CSV reading
    df["property_type_id"] = clean_and_convert_column(df["property_type_id"], "string")

    df["property_sub_type_id"] = clean_and_convert_column(
        df["property_sub_type_id"], "int"
    )
    df["property_sub_type_ar"] = clean_and_convert_column(
        df["property_sub_type_ar"], "string"
    )
    df["property_sub_type_en"] = clean_and_convert_column(
        df["property_sub_type_en"], "string"
    )
    df["property_usage_id"] = clean_and_convert_column(df["property_usage_id"], "int")
    df["property_usage_ar"] = clean_and_convert_column(
        df["property_usage_ar"], "string"
    )
    df["property_usage_en"] = clean_and_convert_column(
        df["property_usage_en"], "string"
    )
    df["building_name_ar"] = clean_and_convert_column(df["building_name_ar"], "string")
    df["building_name_en"] = clean_and_convert_column(df["building_name_en"], "string")
    df["actual_area"] = clean_and_convert_column(df["actual_area"], "float")
    df["db_inserted_at"] = clean_and_convert_column(df["db_inserted_at"], "date")

    # Rename for consistency
    rename_map = {
        "property_usage_id": "property_usage_id_dld",
        "property_usage_ar": "property_usage_ar_dld",
        "property_usage_en": "property_usage_en_dld",
        "building_name_ar": "building_name_ar_dld",
        "building_name_en": "building_name_en_dld",
        "property_id": "masked_property_id",
    }

    for old_col, new_col in rename_map.items():
        if old_col in df.columns:
            df = df.rename(columns={old_col: new_col})

    logger.info(f"Processed {len(df)} property records")
    return df

    def upload_batch_to_collection(collection, records, batch_size, data_type):
    """
    Upload records to MongoDB collection in batches.

    Args:
        collection (pymongo.collection.Collection): MongoDB collection object.
        records (list of dict): List of records (documents) to insert.
        batch_size (int): Number of records to insert per batch.
        data_type (str): Descriptive name of the data type being uploaded (for logging).

    Returns:
        None
    """
    total_records = len(records)
    inserted_count = 0
    failed_batches = []

    logger.info(
        f"Starting batch upload of {total_records} {data_type} records (batch size: {batch_size})"
    )

    for i in range(0, total_records, batch_size):
        batch_num = i // batch_size + 1
        batch = records[i : i + batch_size]

        try:
            for record in batch:
                for key, value in record.items():
                    if not isinstance(value, (list, dict)) and pd.isnull(value):
                        record[key] = None

            result = collection.insert_many(batch, ordered=False)
            inserted_count += len(result.inserted_ids)

            logger.info(
                f"{data_type.capitalize()} Batch {batch_num}: Successfully inserted {len(batch)} records "
                f"({inserted_count}/{total_records} total)"
            )

        except Exception as batch_error:
            logger.error(
                f"{data_type.capitalize()} Batch {batch_num} failed: {str(batch_error)}"
            )
            failed_batches.append(batch_num)
            continue

    # Summary
    if failed_batches:
        logger.warning(
            f"{data_type.capitalize()} upload completed with errors. Failed batches: {failed_batches}"
        )
        logger.info(
            f"Successfully inserted {inserted_count}/{total_records} {data_type} records"
        )
    else:
        logger.info(
            f"✅ Successfully uploaded all {inserted_count} {data_type} records"
        )

    def upload_to_mongodb(
    df_transactions, df_charges, batch_size=100000, clear_existing=True
):
    """
    Upload processed data to MongoDB.

    Args:
        df_transactions (pandas.DataFrame): Processed transactions data.
        df_charges (pandas.DataFrame): Processed service charges data.
        batch_size (int, optional): Number of records to upload per batch. Defaults to 100000.
        clear_existing (bool, optional): Whether to clear existing data before upload. Defaults to True.

    Raises:
        ValueError: If MongoDB connection URL is not configured.
        Exception: For any errors during upload.
    """
    logger.info("Uploading data to MongoDB...")

    if not DEV_MONGO_DB_URL:
        logger.error("DEV_MONGO_DB_URL not found in environment variables")
        raise ValueError("MongoDB connection URL not configured")

    client = MongoClient(DEV_MONGO_DB_URL)

    try:
        dev_db = client["dev_listings_db"]

        # Define collections (only processed data, no raw collections)
        collections = {
            "transactions": dev_db["dld_transactions_all_properties_v2"],
            "charges": dev_db["dld_service_charges"],
        }

        # Clear existing data if requested
        if clear_existing:
            logger.info("Clearing existing data...")
            for name, collection in collections.items():
                delete_result = collection.delete_many({})
                logger.info(
                    f"Deleted {delete_result.deleted_count} existing {name} records"
                )

        # Upload data
        datasets = [
            (df_transactions, collections["transactions"], "processed transactions"),
            (df_charges, collections["charges"], "processed charges"),
        ]

        for df, collection, description in datasets:
            if not df.empty:
                records = df.to_dict(orient="records")
                logger.info(f"Uploading {len(records)} {description} records...")
                upload_batch_to_collection(collection, records, batch_size, description)

        logger.info("✅ Successfully uploaded all data to MongoDB")

    except Exception as e:
        logger.error(f"❌ Error during upload process: {str(e)}")
        raise

    finally:
        client.close()
        logger.info("MongoDB connection closed")
